CREATE TABLE IF NOT EXISTS matters (
    codemat VARCHAR PRIMARY KEY,
    wording VARCHAR,
    coef FLOAT(8)
);


CREATE TABLE IF NOT EXISTS students (
    numstud INT PRIMARY KEY,
    name VARCHAR,
    firstname VARCHAR,
    dateofbirth DATE,
    street VARCHAR,
    postalcod VARCHAR,
    city VARCHAR
);

CREATE TABLE IF NOT EXISTS tests (
    numtest INT PRIMARY KEY,
    testdate DATE,
    place VARCHAR,
    codemat VARCHAR,
    FOREIGN KEY (codemat) REFERENCES matter (codemat)
);


CREATE TABLE IF NOT EXISTS notations (
    numstud INT,
    numtest INT,
    score FLOAT(8),
    PRIMARY KEY(numstud, numtest),
    FOREIGN KEY (numstud) REFERENCES student (numstud),
    FOREIGN KEY (numtest) REFERENCES student (numtest)
);
